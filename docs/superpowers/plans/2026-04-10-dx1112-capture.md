```python
"""
DX capture module — DXGI Desktop Duplication (mss) + Windows Graphics Capture API.

Provides two backends:
  - capture_screen_mss(x, y, w, h)  → BGR ndarray  (primary, works with DX11/12)
  - capture_window_wgc(hwnd)        → BGR ndarray  (window-specific, optional)
"""
import logging
import numpy as np
from typing import Optional

logger = logging.getLogger(__name__)

# ── mss (DXGI Desktop Duplication) ──────────────────────────────────────────
try:
    import mss
    import mss.tools
    MSS_AVAILABLE = True
except ImportError:
    MSS_AVAILABLE = False
    logger.warning("mss not available. Install with: pip install mss")

# ── Windows Graphics Capture API ────────────────────────────────────────────
try:
    import windows_capture
    WGC_AVAILABLE = True
except ImportError:
    WGC_AVAILABLE = False
    logger.warning("windows-capture not available. Install with: pip install windows-capture")


def capture_screen_mss(x: int, y: int, w: int, h: int) -> Optional[np.ndarray]:
    """
    Capture a screen region using DXGI Desktop Duplication via mss.
    Works with DX11/12 games in both windowed and fullscreen exclusive modes.

    Args:
        x, y: top-left corner in screen coordinates
        w, h: width and height in pixels

    Returns:
        BGR numpy array (H, W, 3) or None on failure
    """
    if not MSS_AVAILABLE:
        logger.error("mss not installed — cannot capture screen region")
        return None
    if w <= 0 or h <= 0:
        return None
    try:
        with mss.mss() as sct:
            monitor = {"top": y, "left": x, "width": w, "height": h}
            frame = sct.grab(monitor)
            # mss returns BGRA — drop alpha channel
            img = np.array(frame)[:, :, :3]
            return img
    except Exception as e:
        logger.error(f"mss capture failed: {e}", exc_info=True)
        return None


def capture_window_wgc(hwnd: int) -> Optional[np.ndarray]:
    """
    Capture a window's client area using Windows Graphics Capture API.
    Works with DX11/12 games. Returns None if WGC is unavailable or fails.

    Args:
        hwnd: Windows window handle

    Returns:
        BGR numpy array or None on failure
    """
    if not WGC_AVAILABLE:
        return None
    if not hwnd:
        return None
    try:
        capture = windows_capture.WindowsCapture(hwnd=hwnd)
        frame = capture.capture_frame()
        if frame is None:
            return None
        img = np.array(frame)
        # windows-capture returns BGRA
        if img.ndim == 3 and img.shape[2] == 4:
            img = img[:, :, :3]
        return img
    except Exception as e:
        logger.debug(f"WGC capture failed for hwnd {hwnd}: {e}")
        return None
```

```python
def capture_window_client(hwnd: int) -> Optional[np.ndarray]:
    """
    Capture the client area of a window.
    Priority: WGC (DX11/12 compatible) → mss region → PrintWindow (GDI fallback)
    """
    if not IS_WINDOWS:
        return None

    # 1. Try Windows Graphics Capture API first (works with DX11/12)
    if WGC_AVAILABLE:
        img = capture_window_wgc(hwnd)
        if img is not None:
            return img
        logger.debug("WGC capture failed, falling back to mss region")

    # 2. Try mss with window screen coordinates (DXGI, works with DX11/12)
    info = get_client_origin_and_size(hwnd)
    if info is not None:
        origin_x, origin_y, client_w, client_h = info
        img = capture_screen_mss(origin_x, origin_y, client_w, client_h)
        if img is not None:
            return img
        logger.debug("mss region capture failed, falling back to PrintWindow")

    # 3. Fallback: PrintWindow (GDI — works for non-DX apps)
    try:
        if info is None:
            info = get_client_origin_and_size(hwnd)
        if info is None:
            return None
        _, _, width, height = info

        hwnd_dc = GetDC(hwnd)
        if not hwnd_dc:
            return None
        mem_dc = gdi32.CreateCompatibleDC(hwnd_dc)
        if not mem_dc:
            ReleaseDC(hwnd, hwnd_dc)
            return None
        bmp = gdi32.CreateCompatibleBitmap(hwnd_dc, width, height)
        if not bmp:
            gdi32.DeleteDC(mem_dc)
            ReleaseDC(hwnd, hwnd_dc)
            return None
        old_bmp = gdi32.SelectObject(mem_dc, bmp)
        result = PrintWindow(hwnd, mem_dc, PW_CLIENTONLY)
        if not result:
            gdi32.BitBlt(mem_dc, 0, 0, width, height, hwnd_dc, 0, 0, SRCCOPY)

        bmi = BITMAPINFO()
        bmi.bmiHeader.biSize = ctypes.sizeof(BITMAPINFOHEADER)
        bmi.bmiHeader.biWidth = width
        bmi.bmiHeader.biHeight = -height
        bmi.bmiHeader.biPlanes = 1
        bmi.bmiHeader.biBitCount = 32
        bmi.bmiHeader.biCompression = BI_RGB

        buffer_size = width * height * 4
        buffer = ctypes.create_string_buffer(buffer_size)
        gdi32.GetDIBits(mem_dc, bmp, 0, height, buffer, ctypes.byref(bmi), DIB_RGB_COLORS)

        gdi32.SelectObject(mem_dc, old_bmp)
        gdi32.DeleteObject(bmp)
        gdi32.DeleteDC(mem_dc)
        ReleaseDC(hwnd, hwnd_dc)

        img = np.frombuffer(buffer, dtype=np.uint8).reshape(height, width, 4)
        return img[:, :, :3].copy()
    except Exception as e:
        logger.error(f"PrintWindow fallback failed for hwnd {hwnd}: {e}", exc_info=True)
        return None
```
